import React, { useState, useEffect, useRef, useCallback } from "react";
import { Settings, ChevronLeft, ChevronRight, BookOpen, X, Check, Zap, Moon, Sun, Truck, Coffee, FileDown } from "lucide-react";
import * as XLSX from "xlsx";

// ---------- Reference data (from the original sheet) ----------
const LEAVE_TYPES = [
  { code: "JF", label: "Jour Férié" },
  { code: "RS", label: "Repos Statutaire" },
  { code: "CP", label: "Congé Permanence" },
  { code: "CCP", label: "Congé Climat Pénible" },
  { code: "CA", label: "Congé Annuel" },
  { code: "PE", label: "Permission Exceptionnelle" },
  { code: "RC", label: "Repos Compensateur" },
];

const WORK_TYPES = [
  { code: "AGS", label: "Application graisse silicone", min: 2, max: 2.23 },
  { code: "NL", label: "Nettoyage et lubrification", min: 2, max: 2.1 },
  { code: "VT3", label: "IP : Visite type III", min: 2.1, max: 2.4 },
  { code: "VT2", label: "IP : Visite type II", min: 2, max: 2.2 },
  { code: "VT1", label: "Visite type I", min: 1.8, max: 2 },
  { code: "IR", label: "Inspection caméra IR", min: 1.8, max: 2 },
  { code: "TP", label: "Travaux de préparation", min: 1.93, max: 2.1 },
  { code: "IPI", label: "Travaux IP ou Intervention", min: 2.07, max: 2 },
  { code: "RHT", label: "Réception des nouveaux ouvrages", min: 2, max: 2.27 },
  { code: "RA", label: "IP : Remplacement appareillages", min: 2, max: 2.5 },
  { code: "P", label: "Permanence", min: null, max: 2.5 },
  { code: "TC", label: "Travaux au Centre", min: null, max: 2 },
  { code: "INT", label: "Intervention ou Permanence", min: null, max: 2.37 },
  { code: "TR", label: "Travaux de Réception", min: null, max: 2.3 },
  { code: "F", label: "Formation", min: null, max: "X" },
];

const DAY_TYPE_META = {
  TC: { label: "Travaux au Centre", kind: "work", color: "#F5A623" },
  AUTRE: { label: "Autre travail / déplacement", kind: "work", color: "#22B8AE" },
  RS: { label: "Repos Statutaire", kind: "rest", color: "#4A5A68" },
  JF: { label: "Jour Férié", kind: "rest", color: "#5A6B79" },
  CA: { label: "Congé Annuel", kind: "leave", color: "#7C6FD6" },
  CP: { label: "Congé Permanence", kind: "leave", color: "#8A7BE0" },
  CCP: { label: "Congé Climat Pénible", kind: "leave", color: "#9B8BE8" },
  PE: { label: "Permission Exceptionnelle", kind: "leave", color: "#E5586B" },
  RC: { label: "Repos Compensateur", kind: "rest", color: "#4A5A68" },
};
const DAY_TYPE_ORDER = ["TC", "AUTRE", "RS", "JF", "CA", "CP", "CCP", "PE", "RC"];

const WEEKDAYS_FR = ["Dimanche", "Lundi", "Mardi", "Mercredi", "Jeudi", "Vendredi", "Samedi"];
const MONTHS_FR = ["Janvier","Février","Mars","Avril","Mai","Juin","Juillet","Août","Septembre","Octobre","Novembre","Décembre"];

function daysInMonth(y, m) { return new Date(y, m + 1, 0).getDate(); }
function pad(n) { return String(n).padStart(2, "0"); }
function monthKey(y, m) { return `${y}-${pad(m + 1)}`; }

function emptyDay(dateStr, weekday) {
  return {
    date: dateStr,
    weekday,
    type: "",
    jour1De: "", jour1A: "", jour2De: "", jour2A: "",
    nuit1De: "", nuit1A: "", nuit2De: "", nuit2A: "",
    hsNM: "", hsM: "",
    panier: false,
    travaux: "",
    fraisVoyage: "",
    depl: { pdej: false, dej: false, din: false, dec: false },
    visa: false,
  };
}

function hoursBetween(t1, t2) {
  if (!t1 || !t2) return 0;
  const [h1, m1] = t1.split(":").map(Number);
  const [h2, m2] = t2.split(":").map(Number);
  let d = (h2 * 60 + m2) - (h1 * 60 + m1);
  if (d < 0) d += 24 * 60;
  return Math.round((d / 60) * 100) / 100;
}

export default function App() {
  const [profile, setProfile] = useState({ nom: "", mle: "", classement: "", service: "", codeEtab: "" });
  const [showProfile, setShowProfile] = useState(false);
  const [showLegend, setShowLegend] = useState(false);
  const [today] = useState(new Date());
  const [year, setYear] = useState(today.getFullYear());
  const [month, setMonth] = useState(today.getMonth());
  const [entries, setEntries] = useState({});
  const [loaded, setLoaded] = useState(false);
  const [savedPulse, setSavedPulse] = useState(false);
  const dayRefs = useRef({});
  const saveTimer = useRef(null);

  const mk = monthKey(year, month);

  // ---- load profile once ----
  useEffect(() => {
    (async () => {
      try {
        const r = await window.storage.get("profile", false);
        if (r) setProfile(JSON.parse(r.value));
      } catch (e) { /* no profile yet */ }
    })();
  }, []);

  // ---- load month entries whenever month changes ----
  useEffect(() => {
    setLoaded(false);
    (async () => {
      try {
        const r = await window.storage.get(`month:${mk}`, false);
        if (r) {
          setEntries(JSON.parse(r.value));
        } else {
          const n = daysInMonth(year, month);
          const obj = {};
          for (let d = 1; d <= n; d++) {
            const dt = new Date(year, month, d);
            const dateStr = `${year}-${pad(month + 1)}-${pad(d)}`;
            obj[dateStr] = emptyDay(dateStr, WEEKDAYS_FR[dt.getDay()]);
          }
          setEntries(obj);
        }
      } catch (e) {
        console.error(e);
      } finally {
        setLoaded(true);
      }
    })();
  }, [mk]);

  // ---- debounced save of month entries ----
  useEffect(() => {
    if (!loaded) return;
    if (saveTimer.current) clearTimeout(saveTimer.current);
    saveTimer.current = setTimeout(async () => {
      try {
        const res = await window.storage.set(`month:${mk}`, JSON.stringify(entries), false);
        if (res) {
          setSavedPulse(true);
          setTimeout(() => setSavedPulse(false), 900);
        }
      } catch (e) { console.error("save failed", e); }
    }, 500);
    return () => clearTimeout(saveTimer.current);
  }, [entries, loaded, mk]);

  const saveProfile = useCallback(async (p) => {
    setProfile(p);
    try { await window.storage.set("profile", JSON.stringify(p), false); } catch (e) { console.error(e); }
  }, []);

  function updateDay(dateStr, patch) {
    setEntries((prev) => ({ ...prev, [dateStr]: { ...prev[dateStr], ...patch } }));
  }

  function onTypeChange(dateStr, type) {
    const meta = DAY_TYPE_META[type];
    const patch = { type };
    if (meta && meta.kind !== "work") {
      patch.jour1De = ""; patch.jour1A = ""; patch.jour2De = ""; patch.jour2A = "";
      patch.nuit1De = ""; patch.nuit1A = ""; patch.nuit2De = ""; patch.nuit2A = "";
      patch.hsNM = ""; patch.hsM = "";
      patch.panier = false;
      patch.travaux = meta.label;
    } else if (type === "TC") {
      patch.travaux = entries[dateStr]?.travaux || "Travaux au Centre";
      patch.panier = true;
    }
    updateDay(dateStr, patch);
  }

  const dateList = Object.keys(entries).sort();

  // ---- totals ----
  const totals = dateList.reduce((acc, d) => {
    const e = entries[d];
    const meta = DAY_TYPE_META[e.type];
    if (meta?.kind === "work") acc.worked += 1;
    else if (meta?.kind === "rest") acc.rest += 1;
    else if (meta?.kind === "leave") acc.leave += 1;
    acc.hsNM += Number(e.hsNM) || 0;
    acc.hsM += Number(e.hsM) || 0;
    acc.panier += e.panier ? 1 : 0;
    acc.frais += Number(e.fraisVoyage) || 0;
    acc.pdej += e.depl.pdej ? 1 : 0;
    acc.dej += e.depl.dej ? 1 : 0;
    acc.din += e.depl.din ? 1 : 0;
    acc.dec += e.depl.dec ? 1 : 0;
    return acc;
  }, { worked: 0, rest: 0, leave: 0, hsNM: 0, hsM: 0, panier: 0, frais: 0, pdej: 0, dej: 0, din: 0, dec: 0 });

  function scrollToDay(dateStr) {
    dayRefs.current[dateStr]?.scrollIntoView({ behavior: "smooth", block: "center" });
  }

  function copySummary() {
    const lines = [
      `Feuille de pointage — ${MONTHS_FR[month]} ${year}`,
      profile.nom ? `Nom : ${profile.nom}` : null,
      profile.mle ? `Mle : ${profile.mle}` : null,
      profile.service ? `Service : ${profile.service}` : null,
      "",
      `Jours travaillés : ${totals.worked}`,
      `Repos : ${totals.rest}`,
      `Congés / permissions : ${totals.leave}`,
      `Total HS Normal : ${totals.hsNM}`,
      `Total HS Majoré : ${totals.hsM}`,
      `Paniers : ${totals.panier}`,
      `Frais voyage : ${totals.frais}`,
      `Déplacements — P.Déj:${totals.pdej} Déj:${totals.dej} Dîn:${totals.din} Déc:${totals.dec}`,
    ].filter(Boolean).join("\n");
    navigator.clipboard?.writeText(lines);
  }

  function shiftMonth(delta) {
    let m = month + delta, y = year;
    if (m < 0) { m = 11; y -= 1; }
    if (m > 11) { m = 0; y += 1; }
    setMonth(m); setYear(y);
  }

  function exportOfficialSheet() {
    // ---- Sheet 1 : la feuille de pointage (même disposition que le fichier original) ----
    const rows = [];
    rows.push(["MOIS :", `${MONTHS_FR[month]} ${year}`]);
    rows.push(["Service :", profile.service || ""]);
    rows.push(["Code Etab :", profile.codeEtab || ""]);
    rows.push(["Nom et Prénom :", profile.nom || "", "Mle :", profile.mle || "", "Classement :", profile.classement || ""]);
    rows.push([]);
    // header row 1
    rows.push(["Date", "Jour", "Horaire de travail", "", "", "", "", "", "", "", "Total HS", "", "Panier", "Travaux effectués", "Frais\nvoyage", "Deplacements", "", "", "", "visa chef\nDirect"]);
    // header row 2
    rows.push(["", "", "Jour (5h-21h)", "", "", "", "Nuit (21h-5h)", "", "", "", "", "", "", "", "", "", "", "", "", ""]);
    // header row 3
    rows.push(["", "", "de", "à", "de", "à", "de", "à", "de", "à", "NM", "M", "", "", "", "P.Déj", "Déj", "Din", "Déc", ""]);

    const headerRowsCount = rows.length;

    dateList.forEach((d) => {
      const e = entries[d];
      rows.push([
        e.date, e.weekday,
        e.jour1De, e.jour1A, e.jour2De, e.jour2A,
        e.nuit1De, e.nuit1A, e.nuit2De, e.nuit2A,
        e.hsNM, e.hsM,
        e.panier ? 1 : "",
        e.travaux,
        e.fraisVoyage,
        e.depl.pdej ? 1 : "", e.depl.dej ? 1 : "", e.depl.din ? 1 : "", e.depl.dec ? 1 : "",
        e.visa ? "X" : "",
      ]);
    });

    // totals row
    rows.push([
      "", "", "", "", "", "", "", "", "", "",
      totals.hsNM, totals.hsM, totals.panier, "TOTAL", totals.frais,
      totals.pdej, totals.dej, totals.din, totals.dec, "",
    ]);

    const ws = XLSX.utils.aoa_to_sheet(rows);
    ws["!merges"] = [
      { s: { r: headerRowsCount - 3, c: 2 }, e: { r: headerRowsCount - 3, c: 9 } }, // Horaire de travail
      { s: { r: headerRowsCount - 3, c: 10 }, e: { r: headerRowsCount - 3, c: 11 } }, // Total HS
      { s: { r: headerRowsCount - 3, c: 15 }, e: { r: headerRowsCount - 3, c: 18 } }, // Deplacements
      { s: { r: headerRowsCount - 2, c: 2 }, e: { r: headerRowsCount - 2, c: 5 } }, // Jour (5h-21h)
      { s: { r: headerRowsCount - 2, c: 6 }, e: { r: headerRowsCount - 2, c: 9 } }, // Nuit (21h-5h)
      { s: { r: 3, c: 0 }, e: { r: 3, c: 1 } },
    ];
    ws["!cols"] = [
      { wch: 11 }, { wch: 10 }, { wch: 7 }, { wch: 7 }, { wch: 7 }, { wch: 7 },
      { wch: 7 }, { wch: 7 }, { wch: 7 }, { wch: 7 }, { wch: 6 }, { wch: 6 },
      { wch: 7 }, { wch: 34 }, { wch: 9 }, { wch: 7 }, { wch: 6 }, { wch: 6 }, { wch: 6 }, { wch: 10 },
    ];

    // ---- Sheet 2 : légende / référence (identique à l'original) ----
    const legendRows = [["Code", "Libellé", ""], ...LEAVE_TYPES.map((t) => [t.code, t.label, "X"])];
    legendRows.push([]);
    legendRows.push(["Abréviation", "Type travaux", "Note Min", "Note Max"]);
    WORK_TYPES.forEach((t) => legendRows.push([t.code, t.label, t.min ?? "", t.max ?? ""]));
    const ws2 = XLSX.utils.aoa_to_sheet(legendRows);
    ws2["!cols"] = [{ wch: 12 }, { wch: 34 }, { wch: 10 }, { wch: 10 }];

    const wb = XLSX.utils.book_new();
    XLSX.utils.book_append_sheet(wb, ws, "Feuille de pointage");
    XLSX.utils.book_append_sheet(wb, ws2, "Légende");

    const out = XLSX.write(wb, { bookType: "xlsx", type: "array" });
    const blob = new Blob([out], { type: "application/octet-stream" });
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url;
    a.download = `Feuille_Pointage_${profile.nom ? profile.nom.replace(/\s+/g, "_") + "_" : ""}${mk}.xlsx`;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    URL.revokeObjectURL(url);
  }

  return (
    <div style={{ fontFamily: "'Inter', sans-serif" }} className="min-h-screen bg-[#0D1117] text-[#E7EDF3] pb-10">
      <style>{`
        @import url('https://fonts.googleapis.com/css2?family=Space+Grotesk:wght@500;600;700&family=Inter:wght@400;500;600&family=JetBrains+Mono:wght@500;600&display=swap');
        .disp { font-family: 'Space Grotesk', sans-serif; }
        .mono { font-family: 'JetBrains Mono', monospace; }
        input[type=time]::-webkit-calendar-picker-indicator { filter: invert(1); opacity: 0.5; }
        .scrollbar-none::-webkit-scrollbar { display: none; }
        @keyframes pulse-save { 0% { opacity: 0; } 30% { opacity: 1; } 100% { opacity: 0; } }
        .pulse { animation: pulse-save 0.9s ease; }
      `}</style>

      {/* Header */}
      <div className="sticky top-0 z-20 bg-[#0D1117]/95 backdrop-blur border-b border-[#212B34]">
        <div className="flex items-center justify-between px-4 pt-4 pb-2">
          <div>
            <div className="disp text-lg font-semibold tracking-tight flex items-center gap-2">
              <Zap size={18} className="text-[#F5A623]" strokeWidth={2.5} />
              Feuille de Pointage
            </div>
            <div className="text-xs text-[#7C8B99] mono mt-0.5">
              {profile.nom || "Aucun agent défini"} {profile.mle && `· Mle ${profile.mle}`}
            </div>
          </div>
          <div className="flex items-center gap-1">
            <span className={`text-[10px] text-[#4CD787] mono mr-1 ${savedPulse ? "pulse" : "opacity-0"}`}>enregistré</span>
            <button onClick={() => setShowLegend(true)} className="p-2 rounded-lg bg-[#161D25] border border-[#212B34] active:scale-95 transition">
              <BookOpen size={16} className="text-[#7C8B99]" />
            </button>
            <button onClick={() => setShowProfile(true)} className="p-2 rounded-lg bg-[#161D25] border border-[#212B34] active:scale-95 transition">
              <Settings size={16} className="text-[#7C8B99]" />
            </button>
          </div>
        </div>

        {/* Month nav */}
        <div className="flex items-center justify-between px-4 py-2">
          <button onClick={() => shiftMonth(-1)} className="p-1.5 rounded-md hover:bg-[#161D25]"><ChevronLeft size={18} /></button>
          <div className="disp font-semibold text-sm tracking-wide">{MONTHS_FR[month]} {year}</div>
          <button onClick={() => shiftMonth(1)} className="p-1.5 rounded-md hover:bg-[#161D25]"><ChevronRight size={18} /></button>
        </div>

        {/* Circuit trace strip */}
        <div className="px-3 pb-3 overflow-x-auto scrollbar-none">
          <div className="flex items-center relative" style={{ minWidth: dateList.length * 22 }}>
            <div className="absolute left-0 right-0 h-[2px] bg-[#212B34] top-1/2 -translate-y-1/2" />
            {dateList.map((d) => {
              const e = entries[d];
              const meta = DAY_TYPE_META[e.type];
              const color = meta ? meta.color : "#2A343F";
              const dayNum = Number(d.slice(-2));
              const isToday = d === `${today.getFullYear()}-${pad(today.getMonth()+1)}-${pad(today.getDate())}`;
              return (
                <button key={d} onClick={() => scrollToDay(d)} className="relative flex flex-col items-center shrink-0" style={{ width: 22 }}>
                  <div
                    className="rounded-full transition-all"
                    style={{
                      width: isToday ? 10 : 8, height: isToday ? 10 : 8,
                      background: color, boxShadow: meta ? `0 0 6px ${color}99` : "none",
                      border: isToday ? "2px solid #E7EDF3" : "none",
                    }}
                  />
                  {dayNum % 5 === 0 && <span className="text-[8px] text-[#5A6B79] mono mt-1">{dayNum}</span>}
                </button>
              );
            })}
          </div>
        </div>
      </div>

      {/* Summary bar */}
      <div className="grid grid-cols-4 gap-2 px-4 mt-4 mb-2">
        <SummaryChip label="Travail" value={totals.worked} color="#F5A623" />
        <SummaryChip label="Repos" value={totals.rest} color="#4A5A68" />
        <SummaryChip label="Congés" value={totals.leave} color="#7C6FD6" />
        <SummaryChip label="Panier" value={totals.panier} color="#22B8AE" />
      </div>
      <div className="grid grid-cols-2 gap-2 px-4 mb-4">
        <SummaryChip label="HS Normal" value={totals.hsNM} color="#F5A623" wide />
        <SummaryChip label="HS Majoré" value={totals.hsM} color="#E5586B" wide />
      </div>

      {/* Day cards */}
      <div className="px-4 flex flex-col gap-3">
        {dateList.map((d) => (
          <DayCard
            key={d}
            ref={(el) => (dayRefs.current[d] = el)}
            entry={entries[d]}
            onTypeChange={(t) => onTypeChange(d, t)}
            onChange={(patch) => updateDay(d, patch)}
          />
        ))}
      </div>

      <div className="px-4 mt-6 flex flex-col gap-2">
        <button onClick={exportOfficialSheet} className="w-full py-3 rounded-xl bg-[#F5A623] text-[#0D1117] text-sm font-semibold active:scale-[0.99] transition disp flex items-center justify-center gap-2">
          <FileDown size={16} />
          Exporter la feuille officielle (Excel)
        </button>
        <button onClick={copySummary} className="w-full py-3 rounded-xl bg-[#161D25] border border-[#212B34] text-sm font-medium text-[#E7EDF3] active:scale-[0.99] transition disp">
          Copier le résumé du mois
        </button>
      </div>

      {showProfile && <ProfileModal profile={profile} onSave={saveProfile} onClose={() => setShowProfile(false)} />}
      {showLegend && <LegendModal onClose={() => setShowLegend(false)} />}
    </div>
  );
}

function SummaryChip({ label, value, color, wide }) {
  return (
    <div className="rounded-xl bg-[#161D25] border border-[#212B34] px-3 py-2.5 flex flex-col gap-0.5" style={wide ? {} : {}}>
      <span className="text-[10px] text-[#7C8B99] uppercase tracking-wide disp">{label}</span>
      <span className="mono text-lg font-semibold" style={{ color }}>{value}</span>
    </div>
  );
}

const DayCard = React.forwardRef(function DayCard({ entry, onTypeChange, onChange }, ref) {
  const [expanded, setExpanded] = useState(false);
  const meta = DAY_TYPE_META[entry.type];
  const isWork = !meta || meta.kind === "work";
  const jourH = hoursBetween(entry.jour1De, entry.jour1A) + hoursBetween(entry.jour2De, entry.jour2A);
  const nuitH = hoursBetween(entry.nuit1De, entry.nuit1A) + hoursBetween(entry.nuit2De, entry.nuit2A);
  const isWeekend = entry.weekday === "Samedi" || entry.weekday === "Dimanche";

  return (
    <div ref={ref} className="rounded-2xl bg-[#161D25] border border-[#212B34] overflow-hidden scroll-mt-40">
      <div className="flex items-center justify-between px-4 pt-3 pb-2">
        <div>
          <div className="disp font-semibold text-sm">{entry.date.slice(-2)} <span className="text-[#7C8B99] font-normal">{entry.weekday}</span></div>
        </div>
        {isWeekend && <span className="text-[9px] uppercase tracking-wide text-[#5A6B79] mono">week-end</span>}
      </div>

      {/* Type chips */}
      <div className="flex gap-1.5 px-4 pb-3 overflow-x-auto scrollbar-none">
        {DAY_TYPE_ORDER.map((t) => {
          const m = DAY_TYPE_META[t];
          const active = entry.type === t;
          return (
            <button
              key={t}
              onClick={() => onTypeChange(t)}
              className="shrink-0 px-2.5 py-1 rounded-full text-[11px] font-medium mono border transition"
              style={{
                background: active ? m.color : "transparent",
                borderColor: active ? m.color : "#2A343F",
                color: active ? "#0D1117" : "#7C8B99",
              }}
            >
              {t}
            </button>
          );
        })}
      </div>

      {entry.type && (
        <div className="px-4 pb-4 flex flex-col gap-3 border-t border-[#212B34] pt-3">
          {isWork ? (
            <>
              <TextField label="Travaux effectués" value={entry.travaux} onChange={(v) => onChange({ travaux: v })} autoFocus />

              <button
                onClick={() => setExpanded((v) => !v)}
                className="text-[11px] text-[#7C8B99] disp flex items-center gap-1 self-start mt-1"
              >
                {expanded ? "− Masquer les détails (heures, panier, déplacements...)" : "+ Ajouter détails (heures, panier, déplacements...)"}
              </button>

              {expanded && (
                <div className="flex flex-col gap-3 pt-1 border-t border-[#212B34]">
                  <div>
                    <div className="text-[11px] text-[#7C8B99] mb-1.5 disp flex items-center gap-1"><Sun size={11} /> Jour (5h–21h)</div>
                    <div className="grid grid-cols-2 gap-3">
                      <TimeField label="1ère plage — de" value={entry.jour1De} onChange={(v) => onChange({ jour1De: v })} />
                      <TimeField label="1ère plage — à" value={entry.jour1A} onChange={(v) => onChange({ jour1A: v })} />
                      <TimeField label="2ème plage — de" value={entry.jour2De} onChange={(v) => onChange({ jour2De: v })} />
                      <TimeField label="2ème plage — à" value={entry.jour2A} onChange={(v) => onChange({ jour2A: v })} />
                    </div>
                  </div>
                  <div>
                    <div className="text-[11px] text-[#7C8B99] mb-1.5 disp flex items-center gap-1"><Moon size={11} /> Nuit (21h–5h)</div>
                    <div className="grid grid-cols-2 gap-3">
                      <TimeField label="1ère plage — de" value={entry.nuit1De} onChange={(v) => onChange({ nuit1De: v })} />
                      <TimeField label="1ère plage — à" value={entry.nuit1A} onChange={(v) => onChange({ nuit1A: v })} />
                      <TimeField label="2ème plage — de" value={entry.nuit2De} onChange={(v) => onChange({ nuit2De: v })} />
                      <TimeField label="2ème plage — à" value={entry.nuit2A} onChange={(v) => onChange({ nuit2A: v })} />
                    </div>
                  </div>
                  {(jourH > 0 || nuitH > 0) && (
                    <div className="text-[11px] mono text-[#7C8B99]">
                      Durée jour : <span className="text-[#F5A623]">{jourH}h</span> · Durée nuit : <span className="text-[#22B8AE]">{nuitH}h</span>
                    </div>
                  )}
                  <div className="grid grid-cols-2 gap-3">
                    <NumField label="HS Normal (NM)" value={entry.hsNM} onChange={(v) => onChange({ hsNM: v })} />
                    <NumField label="HS Majoré (M)" value={entry.hsM} onChange={(v) => onChange({ hsM: v })} />
                  </div>
                  <div className="flex items-center justify-between">
                    <span className="text-xs text-[#7C8B99] disp">Panier</span>
                    <Toggle checked={entry.panier} onChange={(v) => onChange({ panier: v })} />
                  </div>
                  <NumField label="Frais de voyage" value={entry.fraisVoyage} onChange={(v) => onChange({ fraisVoyage: v })} prefix="DH" />
                  <div>
                    <div className="text-[11px] text-[#7C8B99] mb-1.5 disp flex items-center gap-1"><Truck size={11} /> Déplacements</div>
                    <div className="flex gap-2">
                      {[["pdej", "P.Déj"], ["dej", "Déj"], ["din", "Dîn"], ["dec", "Déc"]].map(([key, label]) => (
                        <button
                          key={key}
                          onClick={() => onChange({ depl: { ...entry.depl, [key]: !entry.depl[key] } })}
                          className="flex-1 py-1.5 rounded-lg text-[10px] mono font-medium border transition"
                          style={{
                            background: entry.depl[key] ? "#22B8AE" : "transparent",
                            borderColor: entry.depl[key] ? "#22B8AE" : "#2A343F",
                            color: entry.depl[key] ? "#0D1117" : "#7C8B99",
                          }}
                        >
                          {label}
                        </button>
                      ))}
                    </div>
                  </div>
                  <div className="flex items-center justify-between pt-1">
                    <span className="text-xs text-[#7C8B99] disp flex items-center gap-1"><Check size={12} /> Visa chef direct</span>
                    <Toggle checked={entry.visa} onChange={(v) => onChange({ visa: v })} />
                  </div>
                </div>
              )}
            </>
          ) : (
            <div className="text-xs text-[#7C8B99] disp italic">{DAY_TYPE_META[entry.type]?.label}</div>
          )}
        </div>
      )}
    </div>
  );
});

function TimeField({ label, value, onChange, icon }) {
  return (
    <label className="flex flex-col gap-1">
      <span className="text-[10px] text-[#5A6B79] disp flex items-center gap-1">{icon}{label}</span>
      <input
        type="time"
        value={value}
        onChange={(e) => onChange(e.target.value)}
        className="bg-[#0D1117] border border-[#2A343F] rounded-lg px-2 py-1.5 text-sm mono text-[#E7EDF3] focus:outline-none focus:border-[#F5A623]"
      />
    </label>
  );
}

function NumField({ label, value, onChange, prefix }) {
  return (
    <label className="flex flex-col gap-1">
      <span className="text-[10px] text-[#5A6B79] disp">{label}</span>
      <div className="relative">
        <input
          type="number"
          inputMode="decimal"
          value={value}
          onChange={(e) => onChange(e.target.value)}
          placeholder="0"
          className="w-full bg-[#0D1117] border border-[#2A343F] rounded-lg px-2 py-1.5 text-sm mono text-[#E7EDF3] focus:outline-none focus:border-[#F5A623]"
        />
        {prefix && <span className="absolute right-2 top-1/2 -translate-y-1/2 text-[10px] text-[#5A6B79] mono">{prefix}</span>}
      </div>
    </label>
  );
}

function TextField({ label, value, onChange, autoFocus }) {
  return (
    <label className="flex flex-col gap-1">
      <span className="text-[10px] text-[#5A6B79] disp">{label}</span>
      <input
        type="text"
        value={value}
        onChange={(e) => onChange(e.target.value)}
        placeholder="Description..."
        autoFocus={autoFocus}
        className="bg-[#0D1117] border border-[#2A343F] rounded-lg px-2.5 py-1.5 text-sm text-[#E7EDF3] focus:outline-none focus:border-[#F5A623]"
      />
    </label>
  );
}

function Toggle({ checked, onChange }) {
  return (
    <button
      onClick={() => onChange(!checked)}
      className="w-10 h-6 rounded-full relative transition-colors shrink-0"
      style={{ background: checked ? "#4CD787" : "#2A343F" }}
    >
      <div className="w-4 h-4 rounded-full bg-[#0D1117] absolute top-1 transition-all" style={{ left: checked ? 20 : 4 }} />
    </button>
  );
}
